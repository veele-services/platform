deploy@vps:/var/www/veele$ cat /opt/veele/deploy-staging.sh
#!/bin/bash
set -euo pipefail

export HOME="/home/deploy"
export NVM_DIR="/home/deploy/.nvm"
source "$NVM_DIR/nvm.sh"
nvm use 22 >/dev/null
export PATH="$(dirname "$(which node)"):/usr/local/bin:/usr/bin:/bin"
export npm_config_user_agent="pnpm/11.5.0"

PROJECT_SLUG="veele-staging"
APP_DIR="/var/www/veele-staging"
BRANCH="staging"
LOCK_FILE="/home/deploy/veele-staging-deploy.lock"

exec 9>"$LOCK_FILE"

flock -n 9 || {
  echo "Another staging deploy is already running"
  exit 1
}

echo "=== VEELE STAGING DEPLOY START ==="
date

cd "$APP_DIR"

echo "[1/8] Fetch staging branch"
git fetch origin "$BRANCH"
git reset --hard "origin/$BRANCH"

echo "[1b/8] Remove local build output"
rm -rf artifacts/backoffice/.next
rm -rf artifacts/mockup-sandbox/dist

echo "[1c/8] Load staging environment"
set -a
source /opt/veele/secrets/.env.staging
set +a

echo "[2/8] Install dependencies"
CI=true pnpm install --frozen-lockfile --config.dangerouslyAllowAllBuilds=true

echo "[3/8] Typecheck"
pnpm run typecheck

echo "[4/8] Build"
pnpm --filter @workspace/backoffice run build

echo "[5/8] Restart PM2"
pm2 delete "$PROJECT_SLUG" 2>/dev/null || true
pm2 start ecosystem.staging.config.js --update-env
pm2 save

echo "[6/8] Runtime healthcheck"
sleep 3
curl -fsS http://127.0.0.1:3001 > /dev/null

echo "[7/8] Security quick check"
if ps aux | grep -E "kinsing|kdevtmpfsi|xmrig|miner|stratum|syslog-helper|rsyslogd-a" | grep -v grep; then
  echo "Malware indicator found"
  exit 1
fi

echo "[8/8] Done"
echo "=== VEELE STAGING DEPLOY SUCCESS ==="
date
